cmake_minimum_required(VERSION 3.24)
project(imgui_bundle_installed_consumer LANGUAGES CXX)
option(BUNDLE_USE_WCHAR32 "Match an installation built with IMGUI_USE_WCHAR32" OFF)
find_path(BUNDLE_INCLUDE_DIR imgui.h PATH_SUFFIXES include/imgui_bundle REQUIRED)
find_library(BUNDLE_LIBRARY imgui_bundle REQUIRED)
add_executable(bundle_consumer main.cpp)
target_compile_features(bundle_consumer PRIVATE cxx_std_17)
target_include_directories(bundle_consumer PRIVATE "${BUNDLE_INCLUDE_DIR}")
target_compile_definitions(bundle_consumer PRIVATE IMGUI_USER_CONFIG="imconfig_bundle.h" IMGUI_BUNDLE_PYTHON_API)
if(BUNDLE_USE_WCHAR32)
    target_compile_definitions(bundle_consumer PRIVATE IMGUI_USE_WCHAR32)
endif()
target_link_libraries(bundle_consumer PRIVATE "${BUNDLE_LIBRARY}")
enable_testing()
add_test(NAME bundle_installed_consumer COMMAND bundle_consumer)
